import { useState } from "react";
import { Link, useNavigate } from "react-router-dom";
import { Eye, EyeOff, GraduationCap, BookOpen, CheckCircle2 } from "lucide-react";
import { registerMockUser } from "../../services/authService";
import "./Signup.css";
import Logo from "../../Components/common/Logo";

function Signup() {
  const navigate = useNavigate();

  const [form, setForm] = useState({
    name: "",
    email: "",
    password: "",
    confirmPassword: "",
    role: "student",
    acceptTerms: false,
  });

  const [showPassword, setShowPassword] = useState(false);
  const [showConfirmPassword, setShowConfirmPassword] = useState(false);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");
  const [loading, setLoading] = useState(false);

  const updateField = (event) => {
    const { name, value, type, checked } = event.target;

    setForm((current) => ({
      ...current,
      [name]: type === "checkbox" ? checked : value,
    }));

    setError("");
    setSuccess("");
  };

  const handleSubmit = (event) => {
    event.preventDefault();
    setError("");
    setSuccess("");

    const name = form.name.trim();
    const email = form.email.trim().toLowerCase();

    if (name.length < 2) {
      setError("Please enter your full name.");
      return;
    }

    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      setError("Please enter a valid email address.");
      return;
    }

    if (form.password.length < 8) {
      setError("Your password must contain at least 8 characters.");
      return;
    }

    if (!/[A-Za-z]/.test(form.password) || !/\d/.test(form.password)) {
      setError("Use at least one letter and one number in your password.");
      return;
    }

    if (form.password !== form.confirmPassword) {
      setError("Your passwords do not match.");
      return;
    }

    if (!form.acceptTerms) {
      setError("Please accept the terms to continue.");
      return;
    }

    setLoading(true);

    try {
      const result = registerMockUser({
        name,
        email,
        password: form.password,
        role: form.role,
      });

      if (!result.success) {
        setError(result.message);
        return;
      }

      if (result.requiresApproval) {
        setSuccess(
          "Your teacher account has been submitted. An administrator must approve it before you can sign in."
        );
      } else {
        setSuccess("Your account has been created. You can now sign in.");
      }

      setTimeout(() => navigate("/login", { replace: true }), 1800);
    } catch {
      setError("We couldn't create your account. Please try again.");
    } finally {
      setLoading(false);
    }
  };

  return (
    <main className="signup-page">
      <section className="signup-aside">
        <Link
            to="/home"
            className="signup-brand"
            aria-label="OpenNotes home"
            >
            <Logo className="signup-logo" />
        </Link>

        <div className="signup-aside-content">
          <span className="signup-eyebrow">JOIN THE COMMUNITY</span>
          <h1>
            Your next step
            <br />
            <span>starts here.</span>
          </h1>
          <p>
            Discover resources, organise your learning, and make every
            semester count.
          </p>

          <div className="signup-benefit">
            <CheckCircle2 size={19} />
            <span>Study resources organised in one place</span>
          </div>
          <div className="signup-benefit">
            <CheckCircle2 size={19} />
            <span>A profile that keeps your learning connected</span>
          </div>
          <div className="signup-benefit">
            <CheckCircle2 size={19} />
            <span>Different tools for students and teachers</span>
          </div>
        </div>

        <p className="signup-aside-footer">
          Every semester, every subject, right at your fingertips.
        </p>
      </section>

      <section className="signup-main">
        <div className="signup-card">
          <div className="signup-mobile-brand">
            <Logo className="signup-logo" />
          </div>


          <header className="signup-heading">
            <span className="signup-eyebrow">CREATE YOUR ACCOUNT</span>
            <h2>Join OpenNotes</h2>
            <p>Fill in your details to get started.</p>
          </header>

          <form className="signup-form" onSubmit={handleSubmit}>
            <div className="signup-field">
              <label htmlFor="signup-name">Full name</label>
              <input
                id="signup-name"
                name="name"
                type="text"
                autoComplete="name"
                placeholder="Enter your full name"
                value={form.name}
                onChange={updateField}
                maxLength={100}
                required
              />
            </div>

            <div className="signup-field">
              <label htmlFor="signup-email">Email address</label>
              <input
                id="signup-email"
                name="email"
                type="email"
                autoComplete="email"
                placeholder="you@example.com"
                value={form.email}
                onChange={updateField}
                required
              />
            </div>

            <fieldset className="signup-role-fieldset">
              <legend>I want to join as</legend>

              <div className="signup-role-options">
                <label
                  className={`signup-role-option ${
                    form.role === "student" ? "selected" : ""
                  }`}
                >
                  <input
                    type="radio"
                    name="role"
                    value="student"
                    checked={form.role === "student"}
                    onChange={updateField}
                  />
                  <GraduationCap size={21} />
                  <span>
                    <strong>Student</strong>
                    <small>Learn and access resources</small>
                  </span>
                </label>

                <label
                  className={`signup-role-option ${
                    form.role === "teacher" ? "selected" : ""
                  }`}
                >
                  <input
                    type="radio"
                    name="role"
                    value="teacher"
                    checked={form.role === "teacher"}
                    onChange={updateField}
                  />
                  <BookOpen size={21} />
                  <span>
                    <strong>Teacher</strong>
                    <small>Contribute learning materials</small>
                  </span>
                </label>
              </div>

              {form.role === "teacher" && (
                <p className="signup-approval-note">
                  Teacher accounts require administrator approval before
                  access is granted.
                </p>
              )}
            </fieldset>

            <div className="signup-field">
              <label htmlFor="signup-password">Password</label>
              <div className="signup-password-wrap">
                <input
                  id="signup-password"
                  name="password"
                  type={showPassword ? "text" : "password"}
                  autoComplete="new-password"
                  placeholder="At least 8 characters"
                  value={form.password}
                  onChange={updateField}
                  minLength={8}
                  required
                />
                <button
                  type="button"
                  className="signup-password-toggle"
                  onClick={() => setShowPassword((value) => !value)}
                  aria-label={showPassword ? "Hide password" : "Show password"}
                >
                  {showPassword ? <EyeOff size={18} /> : <Eye size={18} />}
                </button>
              </div>
              <small className="signup-field-hint">
                Use at least 8 characters, including a letter and a number.
              </small>
            </div>

            <div className="signup-field">
              <label htmlFor="signup-confirm-password">Confirm password</label>
              <div className="signup-password-wrap">
                <input
                  id="signup-confirm-password"
                  name="confirmPassword"
                  type={showConfirmPassword ? "text" : "password"}
                  autoComplete="new-password"
                  placeholder="Enter your password again"
                  value={form.confirmPassword}
                  onChange={updateField}
                  required
                />
                <button
                  type="button"
                  className="signup-password-toggle"
                  onClick={() =>
                    setShowConfirmPassword((value) => !value)
                  }
                  aria-label={
                    showConfirmPassword ? "Hide password" : "Show password"
                  }
                >
                  {showConfirmPassword ? (
                    <EyeOff size={18} />
                  ) : (
                    <Eye size={18} />
                  )}
                </button>
              </div>
            </div>

            <label className="signup-terms">
              <input
                type="checkbox"
                name="acceptTerms"
                checked={form.acceptTerms}
                onChange={updateField}
              />
              <span>
                I agree to the terms of use and privacy policy.
              </span>
            </label>

            {error && (
              <p className="signup-message signup-error" role="alert">
                {error}
              </p>
            )}

            {success && (
              <p className="signup-message signup-success" role="status">
                {success}
              </p>
            )}

            <button
              className="signup-submit"
              type="submit"
              disabled={loading || Boolean(success)}
            >
              {loading ? "Creating account..." : "Create account"}
              {!loading && <span aria-hidden="true">→</span>}
            </button>
          </form>

          <p className="signup-login-link">
            Already have an account? <Link to="/login">Sign in</Link>
          </p>

          <p className="signup-security-note">
            Account creation is currently running in demo mode. Secure
            production registration will be handled by the backend.
          </p>
        </div>
      </section>
    </main>
  );
}

export default Signup;

========== Frontend/src/services/authService.js ==========


import {
  mockUsers,
  ROLES,
  USER_STATUS,
  TEACHER_APPROVAL_STATUS,
} from "../data/authData";

const REGISTERED_USERS_KEY = "openNotes-registered-users";
const USER_OVERRIDES_KEY = "openNotes-user-overrides";

function readJSON(key, fallback = []) {
  try {
    const raw = localStorage.getItem(key);
    if (!raw) return fallback;

    const parsed = JSON.parse(raw);
    return parsed ?? fallback;
  } catch {
    return fallback;
  }
}

function readRegisteredUsers() {
  const users = readJSON(REGISTERED_USERS_KEY, []);
  return Array.isArray(users) ? users : [];
}

function readOverrides() {
  const overrides = readJSON(USER_OVERRIDES_KEY, []);
  return Array.isArray(overrides) ? overrides : [];
}

function writeJSON(key, value) {
  localStorage.setItem(key, JSON.stringify(value));
}

function safeUser(user) {
  if (!user) return null;

  const { password, ...publicUser } = user;
  return publicUser;
}

function getAllUsers() {
  const allUsers = [...mockUsers, ...readRegisteredUsers()];
  const overrides = readOverrides();

  return allUsers.map((user) => {
    const override = overrides.find(
      (item) => item.email === user.email
    );

    return override
      ? { ...user, ...override }
      : user;
  });
}

function saveUserOverride(email, updates) {
  const normalizedEmail = email.trim().toLowerCase();
  const overrides = readOverrides();

  const existingIndex = overrides.findIndex(
    (item) => item.email === normalizedEmail
  );

  const updatedOverride = {
    ...(existingIndex >= 0 ? overrides[existingIndex] : {}),
    email: normalizedEmail,
    ...updates,
  };

  if (existingIndex >= 0) {
    overrides[existingIndex] = updatedOverride;
  } else {
    overrides.push(updatedOverride);
  }

  writeJSON(USER_OVERRIDES_KEY, overrides);
}

export function registerMockUser({
  name,
  email,
  password,
  role,
}) {
  const normalizedEmail = email.trim().toLowerCase();

  if (![ROLES.STUDENT, ROLES.TEACHER].includes(role)) {
    return {
      success: false,
      message: "This account type cannot be created through public signup.",
    };
  }

  if (
    getAllUsers().some(
      (user) => user.email.trim().toLowerCase() === normalizedEmail
    )
  ) {
    return {
      success: false,
      message: "An account with this email already exists.",
    };
  }

  const isTeacher = role === ROLES.TEACHER;

  const newUser = {
    id:
      typeof crypto !== "undefined" &&
      typeof crypto.randomUUID === "function"
        ? crypto.randomUUID()
        : `user-${Date.now()}-${Math.random().toString(36).slice(2)}`,
    name: name.trim(),
    email: normalizedEmail,
    password,
    role,
    status: isTeacher ? USER_STATUS.PENDING : USER_STATUS.ACTIVE,
    teacherApprovalStatus: isTeacher
      ? TEACHER_APPROVAL_STATUS.PENDING
      : TEACHER_APPROVAL_STATUS.APPROVED,
    createdAt: new Date().toISOString(),
  };

  try {
    writeJSON(REGISTERED_USERS_KEY, [
      ...readRegisteredUsers(),
      newUser,
    ]);
  } catch {
    return {
      success: false,
      message: "Unable to save the demo account in this browser.",
    };
  }

  return {
    success: true,
    message: isTeacher
      ? "Teacher registration submitted for approval."
      : "Account created successfully.",
    requiresApproval: isTeacher,
    user: safeUser(newUser),
  };
}

export function loginMockUser(email, password) {
  const normalizedEmail = email.trim().toLowerCase();

  const user = getAllUsers().find(
    (item) => item.email.trim().toLowerCase() === normalizedEmail
  );

  if (!user || user.password !== password) {
    return {
      success: false,
      message: "Incorrect email or password.",
    };
  }

  if (user.status === USER_STATUS.SUSPENDED) {
    return {
      success: false,
      message: "This account has been suspended. Contact an administrator.",
    };
  }

  if (user.status === USER_STATUS.REJECTED) {
    return {
      success: false,
      message: "This account registration was rejected.",
    };
  }

  if (
    user.role === ROLES.TEACHER &&
    user.teacherApprovalStatus !== TEACHER_APPROVAL_STATUS.APPROVED
  ) {
    return {
      success: false,
      message: "Your teacher account is awaiting administrator approval.",
    };
  }

  if (user.status === USER_STATUS.PENDING) {
    return {
      success: false,
      message: "Your account is awaiting approval.",
    };
  }

  return {
    success: true,
    message: "Signed in successfully.",
    user: safeUser(user),

export function updateTeacherApproval(email, decision) {
  const normalizedEmail = email.trim().toLowerCase();

  if (!["approve", "reject"].includes(decision)) {
    return {
      success: false,
      message: "Invalid approval decision.",
    };
  }

  const user = getAllUsers().find(
    (item) => item.email.trim().toLowerCase() === normalizedEmail
  );

  if (!user || user.role !== ROLES.TEACHER) {
    return {
      success: false,
      message: "Teacher account not found.",
    };
  }

  if (
    user.teacherApprovalStatus !== TEACHER_APPROVAL_STATUS.PENDING
  ) {
    return {
      success: false,
      message: "This teacher request has already been processed.",
    };
  }

  try {
    const approved = decision === "approve";

    saveUserOverride(normalizedEmail, {
      status: approved ? USER_STATUS.ACTIVE : USER_STATUS.REJECTED,
      teacherApprovalStatus: approved
        ? TEACHER_APPROVAL_STATUS.APPROVED
        : TEACHER_APPROVAL_STATUS.REJECTED,
    });

    return {
      success: true,
      message: approved
        ? "Teacher approved successfully."
        : "Teacher registration rejected.",
    };
  } catch {
    return {
      success: false,
      message: "Unable to save the decision in this browser.",
    };
  }
}

export async function apiRequest(path, options = {}) {
  const { headers = {}, body, ...rest } = options;
  const token = getAccessToken();

  const isFormData =
    typeof FormData !== "undefined" && body instanceof FormData;

  const response = await fetch(
    `${API_BASE_URL}${path.startsWith("/") ? path : `/${path}`}`,
    {
      ...rest,
      body,
      headers: {
        ...(body && !isFormData
          ? { "Content-Type": "application/json" }
          : {}),
        ...(token ? { Authorization: `Bearer ${token}` } : {}),
        ...headers,
      },
    }
  );

  let payload;

  try {
    payload = await response.json();
  } catch {
    payload = null;
  }

  if (!response.ok || payload?.success === false) {
    const error = new Error(
      payload?.message || `Request failed (${response.status}).`
    );
    error.status = response.status;
    error.payload = payload;
    throw error;
  }

  return payload;
}

function toQuery(params = {}) {
  const query = new URLSearchParams();

  Object.entries(params).forEach(([key, value]) => {
    if (value !== undefined && value !== null && value !== "") {
      query.set(key, String(value));
    }
  });

  const result = query.toString();
  return result ? `?${result}` : "";
}

  authApi,

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [authLoading, setAuthLoading] = useState(true);

  useEffect(() => {
    let active = true;

    async function restoreSession() {
      const token = getAccessToken();

      if (!token) {
        if (active) setAuthLoading(false);
        return;
      }

      try {
        const result = await authApi.me();

        if (active) {
          setUser(result?.data || null);
        }
      } catch {
        setAccessToken(null);

        if (active) {
          setUser(null);
        }
      } finally {
        if (active) {
          setAuthLoading(false);
        }
      }
    }

    restoreSession();

    return () => {
      active = false;
    };
  }, []);

  const login = useCallback(async (email, password) => {
    try {
      const result = await authApi.login({
        email: email.trim().toLowerCase(),
        password,
      });

      const data = result?.data;

      if (!data?.accessToken || !data?.user) {
        throw new Error("Invalid login response from server.");
      }

      setAccessToken(data.accessToken);
      setUser(data.user);

      return {
        success: true,
        message: result.message || "Signed in successfully.",
        user: data.user,
      };
    } catch (error) {
      return {
        success: false,
        message: error.message || "Unable to sign in.",
      };
    }
  }, []);

  const logout = useCallback(async () => {
    try {
      if (getAccessToken()) {
        await authApi.logoutAll();
      }
    } catch (error) {
      console.warn("Logout request failed:", error.message);
    } finally {
      setAccessToken(null);
      setUser(null);
    }
  }, []);

  const updateUser = useCallback((updates) => {
    setUser((current) =>
      current
        ? {
            ...current,
            ...updates,
            id: current.id,
            _id: current._id,
            role: current.role,
          }
        : current
    );
  }, []);

  const role = user?.role || ROLES.GUEST;
  const permissions = getRolePermissions(role) || {};

  const can = useCallback(
    (permission) => Boolean(permissions[permission]),
    [permissions]
  );

  const hasRole = useCallback(
    (...roles) => roles.includes(role),
    [role]
  );

  const value = useMemo(
    () => ({
      user,
      currentUser: user || guestUser,
      isAuthenticated: Boolean(user),
      authLoading,
      role,
      permissions,
      can,
      hasRole,
      isGuest: role === ROLES.GUEST,
      isStudent: role === ROLES.STUDENT,
      isTeacher: role === ROLES.TEACHER,
      isAdmin: role === ROLES.ADMIN,
      login,
      logout,
      updateUser,
    }),
    [
      user,
      authLoading,
      role,
      permissions,
      can,
      hasRole,
      login,
      logout,
      updateUser,
    ]
  );

  return (
    <AuthContext.Provider value={value}>
      {children}
    </AuthContext.Provider>
  );
}

export default AuthContext;

function createError(message, statusCode) {
  const error = new Error(message);
  error.statusCode = statusCode;
  return error;
}


/* =========================================================
   REGISTER USER
========================================================= */

export async function registerUser({
  name,
  email,
  password,
  collegeId
}) {
  const normalizedEmail = email.trim().toLowerCase();

  const existingUser = await User.exists({
    email: normalizedEmail
  });

  if (existingUser) {
    throw createError(
      "An account with this email already exists.",
      409
    );
  }

  let selectedCollege = null;

  if (collegeId) {
    selectedCollege = await College.findOne({
      _id: collegeId,
      isActive: true,
      verificationStatus: "verified"
    })
      .select("_id")
      .lean();

    if (!selectedCollege) {
      throw createError(
        "The selected college does not exist or is not verified.",
        400
      );
    }
  }

  const passwordHash = await hashPassword(password);

  const {
    token,
    tokenHash,
    expiresAt
  } = generateEmailVerificationToken();

  let user;

  try {
    user = await User.create({
      name,
      email: normalizedEmail,
      passwordHash,

      // Public registration can only create students.
      role: "student",

      collegeId: selectedCollege?._id ?? null,

      collegeAffiliationStatus: selectedCollege
        ? "pending"
        : "not_provided",

      emailVerified: false,

      emailVerificationTokenHash: tokenHash,
      emailVerificationExpiresAt: expiresAt,

      teacherApprovalStatus: "not_applicable",

      authProvider: "local",
      sessionVersion: 0
    });
  } catch (error) {
    if (
      error.code === 11000 &&
      error.keyPattern?.email
    ) {
      throw createError(
        "An account with this email already exists.",
        409
      );
    }

    throw error;
  }

  try {
    await sendVerificationEmail({
      email: user.email,
      token
    });
  } catch (error) {
    console.error(
      "Verification email delivery failed:",
      error.message
    );

    // The account remains unverified.
    // The raw verification token is never returned.
    throw createError(
      "Your account was created, but the verification email could not be sent. Please try again later.",
      503
    );
  }

  return {
    id: user._id,
    name: user.name,
    email: user.email,
    role: user.role,
    collegeId: user.collegeId,
    collegeAffiliationStatus:
      user.collegeAffiliationStatus,
    emailVerified: user.emailVerified,
    message:
      "Registration successful. Please check your email to verify your account."
  };
}

   LOGIN USER
========================================================= */

export async function loginUser({ email, password }) {
  const normalizedEmail = email.trim().toLowerCase();

  const user = await User.findOne({
    email: normalizedEmail
  }).select("+passwordHash +sessionVersion");

  // Use the same error for unknown email and wrong password.
  if (
    !user ||
    !user.passwordHash ||
    !(await verifyPassword(password, user.passwordHash))
  ) {
    throw createError(
      "Invalid email or password.",
      401
    );
  }

  if (!user.isActive) {
    throw createError(
      "This account is inactive.",
      403
    );
  }

  if (!user.emailVerified) {
    const error = createError(
      "Please verify your email address before logging in.",
      403
    );

    error.code = "EMAIL_NOT_VERIFIED";

    throw error;
  }

  // Important: sessionVersion is explicitly selected above.
  // The generated JWT must contain the actual database value.
  const accessToken = await generateAccessToken(user);

  user.lastLoginAt = new Date();

  await user.save();

  return {
    accessToken,
    tokenType: "Bearer",
    expiresIn: 900,

    user: {
      id: user._id,
      name: user.name,
      email: user.email,
      role: user.role,
      collegeId: user.collegeId,
      emailVerified: user.emailVerified,
      collegeAffiliationStatus:
        user.collegeAffiliationStatus,
      teacherApprovalStatus:
        user.teacherApprovalStatus
    }
  };
}

   VERIFY EMAIL
========================================================= */

export async function verifyUserEmail(token) {
  let tokenHash;

  try {
    tokenHash = hashEmailVerificationToken(token);
  } catch {
    throw createError(
      "Invalid or expired verification link.",
      400
    );
  }

  const user = await User.findOne({
    emailVerificationTokenHash: tokenHash,
    emailVerificationExpiresAt: {
      $gt: new Date()
    },
    emailVerified: false,
    isActive: true
  }).select(
    "+emailVerificationTokenHash +emailVerificationExpiresAt"
  );

  if (!user) {
    throw createError(
      "Invalid or expired verification link.",
      400
    );
  }

  user.emailVerified = true;

  user.emailVerificationTokenHash = null;
  user.emailVerificationExpiresAt = null;

  await user.save();

  return {
    email: user.email,
    emailVerified: user.emailVerified
  };
}

      "If an eligible account exists, a verification email will be sent."

  } = generateEmailVerificationToken();

  } catch (error) {

      "Resend verification email delivery failed:",
      error.message
    );

    throw createError(
      "Unable to send the verification email right now. Please try again later.",
      503
    );
  }

  return genericResponse;
}

/* =========================================================
   RESET PASSWORD
========================================================= */

export async function resetPassword({
  token,
  newPassword
}) {
  let tokenHash;

  try {
    tokenHash = hashPasswordResetToken(token);
  } catch {
    throw createError(
      "The password reset link is invalid or has expired.",
      400
    );
  }

  const passwordHash = await hashPassword(newPassword);

  /*
   * Atomic operation:
   *
   * 1. Match the hashed reset token.
   * 2. Check that it has not expired.
   * 3. Check that the account is active and local.
   * 4. Change the password.
   * 5. Remove the reset token.
   * 6. Increment sessionVersion.
   *
   * Only one concurrent request can successfully consume
   * this reset token.
   */

  const updatedUser = await User.findOneAndUpdate(
    {
      passwordResetTokenHash: tokenHash,

      passwordResetExpiresAt: {
        $gt: new Date()
      },

      isActive: true,
      authProvider: "local"
    },
    {
      $set: {
        passwordHash,
        passwordResetTokenHash: null,
        passwordResetExpiresAt: null
      },

      $inc: {
        sessionVersion: 1
      }
    },
    {
      new: true,
      runValidators: true
    }
  ).select("_id");

  if (!updatedUser) {
    throw createError(
      "The password reset link is invalid or has expired.",
      400
    );
  }

  return {
    message:
      "Password reset successfully. All existing sessions have been invalidated. You can now log in."
  };
}


/* =========================================================
   LOGOUT ALL DEVICES
========================================================= */

export async function revokeAllUserSessions(userId) {
  const user = await User.findOneAndUpdate(
    {
      _id: userId,
      isActive: true
    },
    {
      $inc: {
        sessionVersion: 1
      }
    },
    {
      new: true
    }
  ).select("_id");

  if (!user) {
    throw createError(
      "Your account is unavailable. Please log in again.",
      401
    );
  }

  return {
    message:
      "You have been logged out from all devices. Please log in again to continue."
  };
}

import argon2 from "argon2";

const PASSWORD_MIN_LENGTH = 12;
const PASSWORD_MAX_LENGTH = 128;

export function validatePassword(password) {
  if (typeof password !== "string") {
    return {
      valid: false,
      message: "Password must be a string."
    };
  }

  if (password.length < PASSWORD_MIN_LENGTH) {
    return {
      valid: false,
      message: `Password must contain at least ${PASSWORD_MIN_LENGTH} characters.`
    };
  }

  if (password.length > PASSWORD_MAX_LENGTH) {
    return {
      valid: false,
      message: `Password cannot exceed ${PASSWORD_MAX_LENGTH} characters.`
    };
  }

  return { valid: true };
}

export async function hashPassword(password) {
  const validation = validatePassword(password);

  if (!validation.valid) {
    throw new Error(validation.message);
  }

  return argon2.hash(password, {
    type: argon2.argon2id
  });
}

export async function verifyPassword(password, passwordHash) {
  if (
    typeof password !== "string" ||
    typeof passwordHash !== "string" ||
    !passwordHash.startsWith("$argon2id$")
  ) {
    return false;
  }

  try {
    return await argon2.verify(passwordHash, password);
  } catch {
    return false;
  }
}
