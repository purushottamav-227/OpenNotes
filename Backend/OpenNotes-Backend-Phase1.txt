import app from "./app.js";
import env from "./config/env.js";
import {
  connectDatabase,
  disconnectDatabase
} from "./config/database.js";

let server;

async function startServer() {
  try {
    await connectDatabase();

    server = app.listen(env.PORT, () => {
      console.log(`OpenNotes API listening on port ${env.PORT}`);
      console.log(`Environment: ${env.NODE_ENV}`);
    });
  } catch (error) {
    console.error("Failed to start OpenNotes API:", error.message);
    process.exitCode = 1;
    await disconnectDatabase();
  }
}

async function shutdown(signal) {
  console.log(`${signal} received. Shutting down gracefully.`);

  if (server) {
    server.close(async (error) => {
      if (error) {
        console.error("Error closing HTTP server:", error.message);
        process.exitCode = 1;
      }

      try {
        await disconnectDatabase();
      } catch (disconnectError) {
        console.error(
          "Error closing MongoDB connection:",
          disconnectError.message
        );
        process.exitCode = 1;
      }

      process.exit();
    });
  } else {
    await disconnectDatabase();
    process.exit();
  }
}

process.on("SIGTERM", () => shutdown("SIGTERM"));
process.on("SIGINT", () => shutdown("SIGINT"));

startServer();

import mongoose from "mongoose";
import env from "./env.js";

export async function connectDatabase() {
  const mongoUri = env.MONGODB_URI.trim();

  if (!mongoUri) {
    throw new Error("MONGODB_URI is not configured.");
  }

  await mongoose.connect(mongoUri, {
    serverSelectionTimeoutMS: 5000
  });

  console.log("MongoDB connected successfully.");
}

export async function disconnectDatabase() {
  if (mongoose.connection.readyState !== 0) {
    await mongoose.disconnect();
  }
}

export function isDatabaseConnected() {
  return mongoose.connection.readyState === 1;
}

import session from "express-session";
import MongoStore from "connect-mongo";

import env from "./env.js";

const SESSION_MAX_AGE_MS = 24 * 60 * 60 * 1000;

export function createSessionMiddleware() {
  if (!env.MONGODB_URI.trim()) {
    throw new Error(
      "MONGODB_URI must be configured before session middleware can start."
    );
  }

  if (!env.SESSION_SECRET || env.SESSION_SECRET.length < 32) {
    throw new Error(
      "SESSION_SECRET must contain at least 32 characters."
    );
  }

  return session({
    name: "opennotes.sid",

    secret: env.SESSION_SECRET,

    resave: false,

    saveUninitialized: false,

    rolling: true,

    store: MongoStore.create({
      mongoUrl: env.MONGODB_URI,
      collectionName: "opennotes_sessions",
      ttl: SESSION_MAX_AGE_MS / 1000,
      autoRemove: "native"
    }),

    cookie: {
      httpOnly: true,
      secure: env.NODE_ENV === "production",
      sameSite: "lax",
      maxAge: SESSION_MAX_AGE_MS
    }
  });
}

import passport from "passport";
import { Strategy as GoogleStrategy } from "passport-google-oauth20";
import { Strategy as GitHubStrategy } from "passport-github2";

import env from "./env.js";

/*
|--------------------------------------------------------------------------
| GOOGLE OAUTH STRATEGY
|--------------------------------------------------------------------------
*/

if (env.GOOGLE_CLIENT_ID && env.GOOGLE_CLIENT_SECRET) {
  passport.use(
    "google",
    new GoogleStrategy(
      {
        clientID: env.GOOGLE_CLIENT_ID,
        clientSecret: env.GOOGLE_CLIENT_SECRET,
        callbackURL: env.GOOGLE_CALLBACK_URL,
        scope: ["profile", "email"],
        state: true
      },
      (accessToken, refreshToken, profile, done) => {
        try {
          const googleData = profile._json ?? {};

          const email = profile.emails?.find(
            (entry) => entry.verified === true
          )?.value;

          const emailVerified = googleData.email_verified === true;

          if (!profile.id || !email || !emailVerified) {
            return done(null, false, {
              message:
                "A verified Google email address is required to sign in."
            });
          }

          return done(null, {
            provider: "google",
            providerUserId: profile.id,
            name: profile.displayName?.trim() || "",
            email: email.trim().toLowerCase(),
            emailVerified: true
          });
        } catch (error) {
          return done(error);
        }
      }
    )
  );
} else {
  console.warn(
    "Google OAuth is disabled because its client credentials are not configured."
  );
}

/*
|--------------------------------------------------------------------------
| GITHUB OAUTH STRATEGY
|--------------------------------------------------------------------------
*/

if (env.GITHUB_CLIENT_ID && env.GITHUB_CLIENT_SECRET) {
  passport.use(
    "github",
    new GitHubStrategy(
      {
        clientID: env.GITHUB_CLIENT_ID,
        clientSecret: env.GITHUB_CLIENT_SECRET,
        callbackURL: env.GITHUB_CALLBACK_URL,
        scope: ["read:user", "user:email"],
        state: true
      },
      async (accessToken, refreshToken, profile, done) => {
        try {
          if (!profile.id) {
            return done(null, false, {
              message: "Unable to identify your GitHub account."
            });
          }

          /*
           * GitHub's profile email may be missing or unverified.
           * Query its authenticated email endpoint instead.
           */
          const response = await fetch(
            "https://api.github.com/user/emails",
            {
              method: "GET",
              headers: {
                Authorization: `Bearer ${accessToken}`,
                Accept: "application/vnd.github+json",
                "X-GitHub-Api-Version": "2022-11-28",
                "User-Agent": "OpenNotes"
              },
              signal: AbortSignal.timeout(10000)
            }
          );

          if (!response.ok) {
            return done(
              new Error(
                "Unable to verify the email address associated with GitHub."
              )
            );
          }

          const emails = await response.json();

          if (!Array.isArray(emails)) {
            return done(
              new Error("GitHub returned an invalid email response.")
            );
          }

          const verifiedEmail = emails.find(
            (entry) =>
              entry.verified === true && entry.primary === true
          );

          if (!verifiedEmail?.email) {
            return done(null, false, {
              message:
                "A verified primary GitHub email address is required."
            });
          }

          return done(null, {
            provider: "github",
            providerUserId: String(profile.id),
            name:
              profile.displayName?.trim() ||
              profile.username?.trim() ||
              "",
            email: verifiedEmail.email.trim().toLowerCase(),
            emailVerified: true
          });
        } catch (error) {
          return done(error);
        }
      }
    )
  );
} else {
  console.warn(
    "GitHub OAuth is disabled because its client credentials are not configured."
  );
}

export default passport;

export async function registerController(req, res, next) {
  try {
    const validation = registerSchema.safeParse(req.body);

    if (!validation.success) {
      return res.status(400).json({
        success: false,
        message: "Invalid registration details.",
        errors: validation.error.issues.map((issue) => ({
          field: issue.path.join("."),
          message: issue.message
        }))
      });
    }

    const user = await registerUser(validation.data);

    return res.status(201).json({
      success: true,
      message:
        "Registration successful. Email verification is required before account activation.",
      data: user
    });
  } catch (error) {
    return next(error);
  }
}

      success: true,

    return next(error);

    ) {
      return res.status(400).json({

      newPassword

      success: true,
      ...result
    });
  } catch (error) {
    next(error);
  }
}

import User from "../models/User.js";
import College from "../models/College.js";
import { hashPassword } from "../utils/password.js";
import { verifyPassword } from "../utils/password.js";
import { generateAccessToken } from "../utils/token.js";
import { generateEmailVerificationToken } from "../utils/emailverification.js";
import {
  sendVerificationEmail,
  sendPasswordResetEmail
} from "./email.service.js";

import { hashEmailVerificationToken } from "../utils/emailverification.js";

import {
  generatePasswordResetToken,
  hashPasswordResetToken
} from "../utils/passwordReset.js";

export async function registerUser({ name, email, password, collegeId }) {
  const normalizedEmail = email.trim().toLowerCase();

  const existingUser = await User.exists({
    email: normalizedEmail
  });

  if (existingUser) {
    const error = new Error("An account with this email already exists.");
    error.statusCode = 409;
    throw error;
  }

  let selectedCollege = null;

  if (collegeId) {
    selectedCollege = await College.findOne({
      _id: collegeId,
      isActive: true,
      verificationStatus: "verified"
    })
      .select("_id")
      .lean();

    if (!selectedCollege) {
      const error = new Error(
        "The selected college does not exist or is not verified."
      );
      error.statusCode = 400;
      throw error;
    }
  }

  const passwordHash = await hashPassword(password);

  const { token, tokenHash, expiresAt } =
    generateEmailVerificationToken();

  let user;

  try {
    user = await User.create({
      name,
      email: normalizedEmail,
      passwordHash,
      role: "student",
      collegeId: selectedCollege?._id ?? null,
      collegeAffiliationStatus: selectedCollege
        ? "pending"
        : "not_provided",
      emailVerified: false,
      emailVerificationTokenHash: tokenHash,
      emailVerificationExpiresAt: expiresAt,
      teacherApprovalStatus: "not_applicable",
      authProvider: "local"
    });
  } catch (error) {
    // Protect against concurrent registrations using the same email.
    if (error.code === 11000 && error.keyPattern?.email) {
      const duplicateError = new Error(
        "An account with this email already exists."
      );
      duplicateError.statusCode = 409;
      throw duplicateError;
    }

    throw error;
  }

  try {
    await sendVerificationEmail({
      email: user.email,
      token
    });
  } catch (error) {
    // The account remains unverified. Never expose the token in the response.
    console.error(
      "Verification email delivery failed:",
      error.message
    );

    const emailError = new Error(
      "Your account was created, but the verification email could not be sent. Please try again later."
    );

    emailError.statusCode = 503;
    throw emailError;
  }

  return {
    id: user._id,
    name: user.name,
    email: user.email,
    role: user.role,
    collegeId: user.collegeId,
    collegeAffiliationStatus: user.collegeAffiliationStatus,
    emailVerified: user.emailVerified,
    message: "Registration successful. Please check your email to verify your account."
  };
}




export async function loginUser({ email, password }) {
  const normalizedEmail = email.trim().toLowerCase();

  const user = await User.findOne({
    email: normalizedEmail
  }).select("+passwordHash");

  // 1. Validate email and password.
  // Use the same error for a missing account and an incorrect password.
  if (
    !user ||
    !user.passwordHash ||
    !(await verifyPassword(password, user.passwordHash))
  ) {
    const error = new Error("Invalid email or password.");
    error.statusCode = 401;
    throw error;
  }

  // 2. Reject inactive accounts.
  if (!user.isActive) {
    const error = new Error("This account is inactive.");
    error.statusCode = 403;
    throw error;
  }

  // 3. Require email verification before allowing login.
  if (!user.emailVerified) {
    const error = new Error(
      "Please verify your email address before logging in."
    );
    error.statusCode = 403;
    error.code = "EMAIL_NOT_VERIFIED";
    throw error;
  }

  // 4. Generate the access token only after all checks pass.
  const accessToken = await generateAccessToken(user);

  // 5. Update the last login timestamp.
  user.lastLoginAt = new Date();
  await user.save();

  // 6. Return the access token and safe user information.
  return {
    accessToken,
    tokenType: "Bearer",
    expiresIn: 900,
    user: {
      id: user._id,
      name: user.name,
      email: user.email,
      role: user.role,
      collegeId: user.collegeId,
      emailVerified: user.emailVerified,
      collegeAffiliationStatus: user.collegeAffiliationStatus,
      teacherApprovalStatus: user.teacherApprovalStatus
    }
  };
}


export async function verifyUserEmail(token) {
  let tokenHash;

  try {
    tokenHash = hashEmailVerificationToken(token);
  } catch {
    const error = new Error("Invalid or expired verification link.");
    error.statusCode = 400;
    throw error;
  }

  const user = await User.findOne({
    emailVerificationTokenHash: tokenHash,
    emailVerificationExpiresAt: { $gt: new Date() },
    emailVerified: false,
    isActive: true
  }).select("+emailVerificationTokenHash +emailVerificationExpiresAt");

  if (!user) {
    const error = new Error("Invalid or expired verification link.");
    error.statusCode = 400;
    throw error;
  }

  user.emailVerified = true;
  user.emailVerificationTokenHash = null;
  user.emailVerificationExpiresAt = null;

  await user.save();

  return {
    email: user.email,
    emailVerified: user.emailVerified
  };
}


export async function resendVerificationEmail({ email }) {
  const normalizedEmail = email.trim().toLowerCase();

  const user = await User.findOne({
    email: normalizedEmail
  });

  // Use a generic response to avoid revealing account existence.
  const genericResponse = {
    message:
      "If an eligible account exists, a verification email will be sent."
  };

  if (!user || !user.isActive || user.emailVerified) {
    return genericResponse;
  }

  // Generate a fresh verification token.
  const {
    token,
    tokenHash,
    expiresAt
  } = generateEmailVerificationToken();

  // Replace the old token and expiry.
  user.emailVerificationTokenHash = tokenHash;
  user.emailVerificationExpiresAt = expiresAt;

  await user.save();

  try {
    await sendVerificationEmail({
      email: user.email,
      token
    });
  } catch (error) {
    console.error(
      "Resend verification email delivery failed:",
      error.message
    );

    const serviceError = new Error(
      "Unable to send the verification email right now. Please try again later."
    );

    serviceError.statusCode = 503;
    throw serviceError;
  }

  return genericResponse;
}

export async function requestPasswordReset({ email }) {
  const normalizedEmail = email.trim().toLowerCase();

  // Return the same response regardless of whether the account exists.
  const genericResponse = {
    message:
      "If an eligible account exists, password reset instructions will be sent."
  };

  const user = await User.findOne({
    email: normalizedEmail,
    isActive: true,
    authProvider: "local"
  });

  if (!user) {
    return genericResponse;
  }

  const {
    token,
    tokenHash,
    expiresAt
  } = generatePasswordResetToken();

  // Store only the hashed token.
  user.passwordResetTokenHash = tokenHash;
  user.passwordResetExpiresAt = expiresAt;

  await user.save();

  try {
    // Reuse the existing email service with a dedicated reset email.
    await sendPasswordResetEmail({
      email: user.email,
      token
    });
  } catch (error) {
    console.error(
      "Password reset email delivery failed:",
      error.message
    );

    // Clear the token if delivery fails.
    user.passwordResetTokenHash = null;
    user.passwordResetExpiresAt = null;
    await user.save();

    const serviceError = new Error(
      "Unable to send password reset instructions right now. Please try again later."
    );

    serviceError.statusCode = 503;
    throw serviceError;
  }

  return genericResponse;
}


export async function resetPassword({ token, newPassword }) {
  // Hash the token received from the client.
  const tokenHash = hashPasswordResetToken(token);

  // Find the user whose reset token matches and has not expired.
  const user = await User.findOne({
    passwordResetTokenHash: tokenHash,
    passwordResetExpiresAt: { $gt: new Date() },
    isActive: true,
    authProvider: "local"
  }).select("+passwordResetTokenHash +passwordResetExpiresAt");

  if (!user) {
    const error = new Error(
      "The password reset link is invalid or has expired."
    );
    error.statusCode = 400;
    throw error;
  }

  // Validate and securely hash the new password.
  const passwordHash = await hashPassword(newPassword);

  user.passwordHash = passwordHash;

  // Invalidate the reset token after successful use.
  user.passwordResetTokenHash = null;
  user.passwordResetExpiresAt = null;

  await user.save();

  return {
    message: "Password reset successfully. You can now log in."
  };
}

import User from "../models/User.js";
import { verifyAccessToken } from "../utils/token.js";

/**
 * Verify the Bearer token and attach req.auth = { userId, role }.
 *
 * Security properties:
 * - Reads role from MongoDB, not from the JWT payload.
 * - Checks sessionVersion to detect revoked tokens (logout, password change).
 * - Rejects inactive accounts.
 */
export async function authenticateUser(req, res, next) {
  try {
    // 1. Read the Authorization header.
    const authorization = req.headers.authorization;

    if (!authorization || !authorization.startsWith("Bearer ")) {
      return res.status(401).json({
        success: false,
        message: "Authentication required."
      });
    }

    // 2. Extract the access token.
    const token = authorization.slice("Bearer ".length).trim();

    if (!token) {
      return res.status(401).json({
        success: false,
        message: "Authentication required."
      });
    }

    // 3. Verify the token's signature and expiration.
    const payload = await verifyAccessToken(token);

    if (
      typeof payload.sub !== "string" ||
      !["student", "teacher", "admin"].includes(payload.role)
    ) {
      return res.status(401).json({
        success: false,
        message: "Invalid authentication token."
      });
    }

    // 4. Fetch the current user from MongoDB to get the live sessionVersion.
    const user = await User.findById(payload.sub)
      .select("_id role isActive sessionVersion");

    // 5. Reject deleted or inactive accounts.
    if (!user || !user.isActive) {
      return res.status(401).json({
        success: false,
        message: "Your account is unavailable. Please log in again."
      });
    }

    // 6. Reject tokens issued before the last session invalidation.
    //    If sv (sessionVersion in JWT) < current sessionVersion, the token is revoked.
    const tokenSv = typeof payload.sv === "number" ? payload.sv : 0;
    if (tokenSv < (user.sessionVersion ?? 0)) {
      return res.status(401).json({
        success: false,
        message: "Your session has expired. Please log in again."
      });
    }

    // 7. Store the current database role, not the token's role.
    req.auth = {
      userId: user._id.toString(),
      role: user.role
    };

    return next();
  } catch (error) {
    // Invalid or expired tokens return 401.
    // Unexpected database errors propagate to the global handler.
    if (
      error?.code === "ERR_JWT_EXPIRED" ||
      error?.code === "ERR_JWS_INVALID" ||
      error?.code === "ERR_JWT_INVALID" ||
      error?.code === "ERR_JWT_CLAIM_VALIDATION_FAILED" ||
      error?.code === "ERR_JWS_SIGNATURE_VERIFICATION_FAILED" ||
      error?.name === "JWTExpired" ||
      error?.name === "JWTClaimValidationFailed" ||
      error?.name === "JWSSignatureVerificationFailed" ||
      error?.name === "JWSInvalid" ||
      error?.name === "JWTInvalid" ||
      error?.name === "CastError"
    ) {
      return res.status(401).json({
        success: false,
        message: "Invalid or expired authentication token."
      });
    }

    return next(error);
  }
}

/**
 * Restrict access to routes based on the authenticated user's role.
 * Must be used after authenticateUser.
 */
export function requireRole(...allowedRoles) {
  return function roleAuthorization(req, res, next) {
    if (!req.auth) {
      return res.status(401).json({
        success: false,
        message: "Authentication required."
      });
    }

    if (!allowedRoles.includes(req.auth.role)) {
      return res.status(403).json({
        success: false,
        message: "You do not have permission to access this resource."
      });
    }

    return next();
  };
}

export async function searchCollegesController(req, res, next) {

    const validation = collegeSearchSchema.safeParse(req.query);

    if (!validation.success) {
      return res.status(400).json({
        success: false,
        message: "Invalid college search parameters.",
        errors: validation.error.issues.map((issue) => ({
          field: issue.path.join("."),
          message: issue.message
        }))
      });
    }

    const { q, limit } = validation.data;
    const colleges = await searchColleges(q, limit);

    return res.status(200).json({
      success: true,
      count: colleges.length,
      data: colleges
    });
  } catch (error) {
    return next(error);
  }
}

export async function getCollegeController(req, res, next) {
  try {
    const validation = collegeIdSchema.safeParse(req.params.id);

    if (!validation.success) {
      return res.status(400).json({
        success: false,
        message: "Invalid college ID."
      });
    }

    const college = await getCollegeById(validation.data);

    if (!college) {
      return res.status(404).json({
        success: false,
        message: "College not found."
      });
    }

    return res.status(200).json({
      success: true,
      data: college
    });
  } catch (error) {
    return next(error);
  }
}

import College from "../models/College.js";

export async function searchColleges(query, limit = 10) {
  const searchTerm = query.trim();

  if (!searchTerm) {
    return [];
  }

  return College.find({
    isActive: true,
    verificationStatus: "verified",
    name: {
      $regex: escapeRegex(searchTerm),
      $options: "i"
    }
  })
    .select("name address website")
    .sort({ name: 1 })
    .limit(limit)
    .lean();
}

export async function getCollegeById(collegeId) {
  return College.findOne({
    _id: collegeId,
    isActive: true,
    verificationStatus: "verified"
  })
    .select("name address website")
    .lean();
}

function escapeRegex(value) {
  return value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
}

import { describe, it, expect } from "vitest";
import {
  validatePassword,
  hashPassword,
  verifyPassword
} from "../src/utils/password.js";

describe("Password security", () => {
  it("accepts a valid password", () => {
    expect(
      validatePassword("SecurePassword123!").valid
    ).toBe(true);
  });

  it("rejects a password shorter than 12 characters", () => {
    const result = validatePassword("Short123!");

    expect(result.valid).toBe(false);
    expect(result.message).toContain("at least 12 characters");
  });

  it("rejects a password longer than 128 characters", () => {
    const longPassword = "A".repeat(129);
    const result = validatePassword(longPassword);

    expect(result.valid).toBe(false);
    expect(result.message).toContain("128 characters");
  });

  it("rejects non-string passwords", () => {
    const result = validatePassword(123456789012);

    expect(result.valid).toBe(false);
    expect(result.message).toBe("Password must be a string.");
  });

  it("creates an Argon2id password hash", async () => {
    const password = "SecurePassword123!";
    const passwordHash = await hashPassword(password);

    expect(passwordHash).toMatch(/^\$argon2id\$/);
    expect(passwordHash).not.toBe(password);
  });

  it("accepts the correct password", async () => {
    const password = "SecurePassword123!";
    const passwordHash = await hashPassword(password);

    await expect(
      verifyPassword(password, passwordHash)
    ).resolves.toBe(true);
  });

  it("rejects an incorrect password", async () => {
    const password = "SecurePassword123!";
    const passwordHash = await hashPassword(password);

    await expect(
      verifyPassword("WrongPassword123!", passwordHash)
    ).resolves.toBe(false);
  });

  it("rejects an invalid password hash", async () => {
    await expect(
      verifyPassword("SecurePassword123!", "invalid-hash")
    ).resolves.toBe(false);
  });

  it("rejects a password that is too short when hashing", async () => {
    await expect(
      hashPassword("Short123!")
    ).rejects.toThrow("Password must contain at least 12 characters.");
  });
});

import { describe, it, expect, beforeEach, afterEach, vi } from "vitest";
import express from "express";
import request from "supertest";
import { rateLimit } from "express-rate-limit";

describe("Authentication rate limiting", () => {
  let app;

  beforeEach(() => {
    vi.useRealTimers();

    app = express();
    app.use(express.json());

    // A small test-only limit makes it possible to verify
    // rate limiting without making many requests.
    const testLimiter = rateLimit({
      windowMs: 60_000,
      limit: 2,
      standardHeaders: "draft-8",
      legacyHeaders: false,
      message: {
        success: false,
        message: "Too many requests."
      }
    });

    app.post("/test-login", testLimiter, (req, res) => {
      res.status(200).json({
        success: true
      });
    });
  });

  afterEach(() => {
    vi.useRealTimers();
  });

  it("allows requests within the limit", async () => {
    const first = await request(app).post("/test-login");
    const second = await request(app).post("/test-login");

    expect(first.status).toBe(200);
    expect(second.status).toBe(200);
  });

  it("blocks requests after the limit is exceeded", async () => {
    await request(app).post("/test-login");
    await request(app).post("/test-login");

    const third = await request(app).post("/test-login");

    expect(third.status).toBe(429);
    expect(third.body.success).toBe(false);
    expect(third.body.message).toBe("Too many requests.");
  });
});
